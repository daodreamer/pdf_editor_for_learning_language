cmake_minimum_required(VERSION 3.24)
project(Langlern VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.8.3 REQUIRED COMPONENTS Core Gui Quick Qml QuickControls2 QuickDialogs2 Sql)
qt_standard_project_setup(REQUIRES 6.8)
configure_file(assets/fonts/NotoSansSC-Regular.ttf fonts/NotoSansSC-Regular.ttf COPYONLY)
configure_file(assets/fonts/OFL.txt fonts/OFL.txt COPYONLY)
if(EXISTS "${PROJECT_SOURCE_DIR}/.deps/dictionary/ecdict.sqlite")
    configure_file(.deps/dictionary/ecdict.sqlite dictionary/ecdict.sqlite COPYONLY)
    configure_file(third_party/notices/ECDICT-MIT.txt dictionary/LICENSE.txt COPYONLY)
endif()
include(CTest)
option(LANGLERN_BUILD_PHASE0 "Build the retained PDFium technology probes" OFF)
function(langlern_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()
if(BUILD_TESTING OR LANGLERN_BUILD_PHASE0)
    find_package(Qt6 6.8.3 REQUIRED COMPONENTS Test)
endif()
find_package(PDFium CONFIG REQUIRED)
if(NOT PDFium_VERSION STREQUAL "157.0.8100.0")
    message(FATAL_ERROR "Expected PDFium 157.0.8100.0, got ${PDFium_VERSION}")
endif()
add_subdirectory(src/pdf)
add_subdirectory(qml/Langlern/Shell)
add_subdirectory(src/app)
if(BUILD_TESTING)
    add_subdirectory(tests/shell)
endif()
if(LANGLERN_BUILD_PHASE0)
    add_subdirectory(prototypes/phase0)
endif()
